cmake_minimum_required(VERSION 3.16)
project(lhatove_extension_probe C)

set(LHAT_INCLUDE_DIR "" CACHE PATH "L^ source include directory")
set(LHAT_GENERATED_INCLUDE_DIR "" CACHE PATH "L^ build include directory (lhat/version.h)")
set(SIGNATURE_HEADER "" CACHE FILEPATH "Embedded signature table generated by the test")
set(PEER_SIGNATURE_HEADER "" CACHE FILEPATH "Peer table, including its dependency on probe")

foreach(target probe peer badabi badversion badsignatures)
    add_library(${target} SHARED probe.c)
    target_include_directories(${target} PRIVATE "${LHAT_GENERATED_INCLUDE_DIR}" "${LHAT_INCLUDE_DIR}")
    set_target_properties(${target} PROPERTIES PREFIX "" C_STANDARD 11 C_VISIBILITY_PRESET hidden)
    if(target STREQUAL "peer" AND PEER_SIGNATURE_HEADER)
        target_compile_definitions(${target} PRIVATE PROBE_SIGNATURE_HEADER="${PEER_SIGNATURE_HEADER}")
    elseif(SIGNATURE_HEADER)
        target_compile_definitions(${target} PRIVATE PROBE_SIGNATURE_HEADER="${SIGNATURE_HEADER}")
    endif()
endforeach()
target_compile_definitions(peer PRIVATE PROBE_PEER)
target_compile_definitions(badabi PRIVATE PROBE_BAD_ABI)
target_compile_definitions(badversion PRIVATE PROBE_BAD_VERSION)
target_compile_definitions(badsignatures PRIVATE PROBE_BAD_SIGNATURES)
add_library(noentry SHARED noentry.c)
set_target_properties(noentry PROPERTIES PREFIX "")
